CREATE DATABASE PAASAL_RIYA_DB_01;
USE PAASAL_RIYA_DB_01;

CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    fullname VARCHAR(100) NOT NULL,
    mobile VARCHAR(15) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    nic VARCHAR(20) NOT NULL UNIQUE,
    district VARCHAR(100) NOT NULL,
    province VARCHAR(100) NOT NULL,
    address TEXT NOT NULL,
    user_type VARCHAR(50) NOT NULL,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE posts (
    post_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    name VARCHAR(100) NOT NULL,
    reg_no VARCHAR(50) NOT NULL,
    province VARCHAR(100) NOT NULL,
    district VARCHAR(100) NOT NULL,
    address TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);




-- ප්‍රධාන සේවා වගුව
CREATE TABLE services (
    service_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    service_name VARCHAR(255) NOT NULL,
    reg_no VARCHAR(50) NOT NULL UNIQUE,
    vehicle_type ENUM('Bus', 'Van', 'Three-wheeler') NOT NULL,
    owner VARCHAR(100) NOT NULL,
    driver VARCHAR(100) NOT NULL,
    driver_reg_no VARCHAR(50) NOT NULL,
    province VARCHAR(100) NOT NULL,
    district VARCHAR(100) NOT NULL,
    address TEXT NOT NULL,
    description TEXT,
    -- Schedule
    has_morning BOOLEAN DEFAULT FALSE,
    morning_place VARCHAR(255),
    morning_time TIME,
    has_evening BOOLEAN DEFAULT FALSE,
    evening_place VARCHAR(255),
    evening_time TIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

-- Assistants (Multiple)
CREATE TABLE service_assistants (
    id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT NOT NULL,
    assistant_name VARCHAR(100) NOT NULL,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

-- Schools (Multiple)
CREATE TABLE service_schools (
    id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT NOT NULL,
    school_name VARCHAR(255) NOT NULL,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

-- Telephone Numbers (Multiple)
CREATE TABLE service_telephones (
    id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT NOT NULL,
    telephone VARCHAR(20) NOT NULL,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

-- Emails (Multiple)
CREATE TABLE service_emails (
    id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT NOT NULL,
    email VARCHAR(100) NOT NULL,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

-- Websites (Multiple)
CREATE TABLE service_websites (
    id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT NOT NULL,
    website VARCHAR(255) NOT NULL,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

-- Favorites table (user_id, service_id) composite primary key
CREATE TABLE favorites (
    user_id INT NOT NULL,
    service_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, service_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

CREATE TABLE ratings (
    rating_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    service_id INT NOT NULL,
    rating TINYINT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY unique_user_service (user_id, service_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE
);

CREATE TABLE comments (
    comment_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    service_id INT NOT NULL,
    comment TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE,
    INDEX idx_service (service_id)
);

CREATE TABLE service_images (
    image_id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT NOT NULL,
    image_path VARCHAR(500) NOT NULL,
    is_mandatory TINYINT NOT NULL COMMENT '1=Front,2=Back,3=Left,4=Right,5=Seats,0=Optional',
    display_order INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (service_id) REFERENCES services(service_id) ON DELETE CASCADE,
    INDEX idx_service (service_id)
);

-- Admin / User messaging table එක
CREATE TABLE post_messages (
    message_id INT AUTO_INCREMENT PRIMARY KEY,
    post_id INT NOT NULL,
    sender_id INT NOT NULL,   -- message එක යවන user
    receiver_id INT NOT NULL, -- message ලබන user (admin හෝ owner)
    message TEXT NOT NULL,
    is_read TINYINT(1) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES services(service_id) ON DELETE CASCADE,
    FOREIGN KEY (sender_id) REFERENCES users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (receiver_id) REFERENCES users(user_id) ON DELETE CASCADE,
    INDEX idx_post (post_id)
);

CREATE TABLE audit_log (
    id INT AUTO_INCREMENT PRIMARY KEY,
    table_name VARCHAR(50) NOT NULL COMMENT 'වෙනස් වූ table එකේ නම',
    record_id INT NOT NULL COMMENT 'බලපෑමට ලක්වූ record එකේ ID',
    action ENUM('INSERT','UPDATE','DELETE') NOT NULL COMMENT 'සිදුකළ ක්‍රියාව',
    old_values JSON NULL COMMENT 'පැරණි දත්ත (JSON format)',
    new_values JSON NULL COMMENT 'නව දත්ත (JSON format)',
    changed_by INT NULL COMMENT 'ක්‍රියාව සිදුකළ user ID (NULL = system/public)',
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'වෙනස් වූ වේලාව',
    ip_address VARCHAR(45) NULL COMMENT 'IP ලිපිනය'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Performance සඳහා indexes
CREATE INDEX idx_audit_table_record ON audit_log(table_name, record_id);
CREATE INDEX idx_audit_changed_by ON audit_log(changed_by);
CREATE INDEX idx_audit_changed_at ON audit_log(changed_at);
